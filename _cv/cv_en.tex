\section{Research Experience}
\entry{\href{https://larsph.github.io/avatarmix/}{AvatarMix: High-Quality 3D Virtual Try-On}}{Doctoral research}
\field{Overview}{Developed a high-quality 3D virtual try-on method that combines two 3D avatars to show one person wearing the other's clothing while preserving the wearer's face and body shape.}
\field{Method}{Proposed a method that combines the wearer's head with the clothing donor's avatar and adapts the entire body to the wearer's shape. Used a diffusion model to repair composition seams and appearance artifacts caused by deformation.}
\field{Contribution}{Formulated the body adaptation and seam repair problems and led method development, implementation, evaluation, visualization, and manuscript preparation.}
\field{Publication}{Accepted to CVPR 2026 Findings.}
\link{https://openaccess.thecvf.com/content/CVPR2026F/html/Wang_AvatarMix_Identity-Preserving_Cross-Avatar_Composition_for_Outfit_Personalization_CVPRF_2026_paper.html}{Paper}\quad \link{https://larsph.github.io/avatarmix/}{Project page}\quad \link{https://github.com/LarsPh/AvatarMix-Code}{Code}

\entry{\href{https://doi.org/10.26599/CVM.2025.9450508}{EG-HumanNeRF: Fast Human Novel-View Synthesis}}{Master's research}
\field{Overview}{Developed a feed-forward method that synthesizes novel views of a person from three or four images, balancing rendering speed and image quality without per-person retraining.}
\field{Method}{Used the SMPL-X human body model and predicted geometry to narrow NeRF sampling regions in two stages, reducing computation. Improved image quality through occlusion-aware image refinement.}
\field{Contribution}{Developed the two-stage sampling method and occlusion-aware refinement, accelerated the overall pipeline, and led implementation, evaluation, and manuscript preparation.}
\field{Publication}{Published in Computational Visual Media, 12(2), 355--379 (2026).}
\link{https://doi.org/10.26599/CVM.2025.9450508}{Paper}\quad \link{https://ieeexplore.ieee.org/document/11373444/}{Publisher}\quad \link{https://github.com/LarsPh/EG-HumanNeRF}{Code}
